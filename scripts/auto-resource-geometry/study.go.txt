func geometryWrite(t *testing.T, p string, v any) {

func geometryEnvelope(points []raytracer.Point, radius float64) raytracer.Bounds {
	b := raytracer.Bounds{MinLon: 180, MinLat: 90, MaxLon: -180, MaxLat: -90}
	for _, p := range points {
		x := raytracer.BoundsAroundPoint(p, radius)
		b.MinLon = math.Min(b.MinLon, x.MinLon)
		b.MinLat = math.Min(b.MinLat, x.MinLat)
		b.MaxLon = math.Max(b.MaxLon, x.MaxLon)
		b.MaxLat = math.Max(b.MaxLat, x.MaxLat)
	}
	return b
}
func geometryDescribe(idx *raytracer.BuildingIndex, b raytracer.Bounds) geometryStats {
	found := idx.SearchBounds(b)
	v := make([]int, 0, len(found))
	logical := map[string]bool{}
	s := geometryStats{Footprints: len(found), Bounds: b}
	for _, f := range found {
		n := len(f.Vertices)
		s.Vertices += n
		e := n
		if n > 1 && f.Vertices[0] == f.Vertices[n-1] {
			e--
		}
		s.Edges += e
		if n > s.Max {
			s.Max = n
		}
		v = append(v, n)
		id := f.LogicalID
		if id == "" {
			id = f.ID
		}
		logical[id] = true
	}
	s.Logical = len(logical)
	sort.Ints(v)
	if len(v) > 0 {
		s.Median = float64(v[len(v)/2])
		if len(v)%2 == 0 {
			s.Median = (float64(v[len(v)/2-1]) + s.Median) / 2
		}
	}
	s.Area = (b.MaxLon - b.MinLon) * 111320 * math.Cos((b.MinLat+b.MaxLat)*math.Pi/360) * (b.MaxLat - b.MinLat) * 111320 / 1e6
	if s.Area > 0 {
		s.Density = float64(s.Footprints) / s.Area
		s.VertexDensity = float64(s.Vertices) / s.Area
	}
	return s
}
func geometryPoints(towers []geometryTower) []raytracer.Point {
	p := []raytracer.Point{}
	for _, x := range towers {
		p = append(p, raytracer.Point{Lon: x.Coordinates[0], Lat: x.Coordinates[1]})
	}
	return p
}
func geometryOverlap(a, b raytracer.Bounds) bool {
	return a.MinLon <= b.MaxLon && a.MaxLon >= b.MinLon && a.MinLat <= b.MaxLat && a.MaxLat >= b.MinLat
}
func TestGeometryDomainSelection(t *testing.T) {
	if os.Getenv("ATOM_GEOMETRY_STUDY") != "1" {
		t.Skip("opt-in disposable calibration")
	}
	w := os.Getenv("ATOM_GEOMETRY_DIR")
	pack, e := raytracer.LoadDatasetPack("../data-pipeline")
	if e != nil {
		t.Fatal(e)
	}
	var candidates []geometryDomain
	geometryRead(t, filepath.Join(w, "candidates.json"), &candidates)
	for i := range candidates {
		p := geometryPoints(candidates[i].Towers)
		candidates[i].Stats = geometryDescribe(pack.BuildingIndex, geometryEnvelope(p, 400))
		candidates[i].SelectionBounds = geometryEnvelope(p, 800)
	}
	sort.Slice(candidates, func(i, j int) bool {
		a, b := candidates[i], candidates[j]
		if a.Stats.VertexDensity != b.Stats.VertexDensity {
			return a.Stats.VertexDensity < b.Stats.VertexDensity
		}
		return a.ID < b.ID
	})
	for i := range candidates {
		candidates[i].Rank = i
	}
	chosen := []geometryDomain{}
	for _, x := range []struct {
		band string
		q    float64
	}{{"very-dense", .98}, {"dense", .85}, {"medium", .50}, {"sparse", .05}} {
		for _, split := range []string{"calibration", "held-out"} {
			target := int(x.q * float64(len(candidates)-1))
			order := append([]geometryDomain(nil), candidates...)
			sort.SliceStable(order, func(i, j int) bool { return absGeometry(order[i].Rank-target) < absGeometry(order[j].Rank-target) })
			ok := false
			for _, c := range order {
				overlap := false
				for _, d := range chosen {
					if geometryOverlap(c.SelectionBounds, d.SelectionBounds) {
						overlap = true
						break
					}
				}
				if !overlap {
					c.Band = x.band
					c.Split = split
					c.ID = x.band + "-" + split
					candidates[c.Rank].Band = x.band
					candidates[c.Rank].Split = split
					chosen = append(chosen, c)
					ok = true
					break
				}
			}
			if !ok {
				t.Fatal("insufficient disjoint domains")
			}
		}
	}
	geometryWrite(t, filepath.Join(w, "domains.json"), map[string]any{"method": "unique inventory-coordinate anchors; nearest six distinct Cell IDs by cosine-adjusted coordinate distance; rank 400m enclosing-box vertex density, ID tie break; target quantiles .05/.50/.85/.98; choose nearest rank with disjoint 800m enclosing boxes; train then holdout per band BEFORE timing", "candidates": candidates, "selected": chosen, "dataset": pack.Manifest, "global_footprints": pack.BuildingIndex.Len(), "global_vertices": pack.BuildingIndex.VertexCount(), "inventory_cells": len(pack.Towers)})
}
func absGeometry(n int) int {
	if n < 0 {
		return -n
	}
	return n
}
func geometryFields(p string) map[string]uint64 {
	b, _ := os.ReadFile(p)
	m := map[string]uint64{}
	for _, line := range strings.Split(string(b), "\n") {
		f := strings.Fields(line)
		if len(f) > 1 {
			m[strings.TrimSuffix(f[0], ":")], _ = strconv.ParseUint(f[1], 10, 64)
		}
	}
	return m
}
func geometryNumber(p string) uint64 {
	b, _ := os.ReadFile(p)
	n, _ := strconv.ParseUint(strings.TrimSpace(string(b)), 10, 64)
	return n
}

						var before, after runtime.MemStats
						runtime.ReadMemStats(&before)
						cpu := capacityCPU()
						start := time.Now()
						var s geometryStats
						count := 0
						for k := 0; k < 100; k++ {
							if kind == "bbox-count" {
								count = len(pack.BuildingIndex.SearchBounds(b))
							} else {
								s = geometryDescribe(pack.BuildingIndex, b)
								count = s.Footprints
							}
						}
						wall := time.Since(start).Seconds() / 100
						cpu = (capacityCPU() - cpu) / 100
						runtime.ReadMemStats(&after)
						rows = append(rows, map[string]any{"domain": d.ID, "radius": r, "cells": n, "kind": kind, "repeat": repeat, "wall_seconds": wall, "cpu_seconds": cpu, "allocation_bytes": float64(after.TotalAlloc-before.TotalAlloc) / 100, "footprints": count, "stats": s})
					}
				}
			}
		}
	}
	geometryWrite(t, filepath.Join(w, "preflight-"+os.Getenv("ATOM_GEOMETRY_PROFILE")+".json"), rows)
}

			}
			if snapshot.CacheHit || snapshot.CompletedRuns != 16 {
				t.Fatal("expected uncached 16-run job")
			}
			job = &snapshot
		case "cancel":
			ctx, cancel := context.WithCancel(context.Background())
			timer := time.AfterFunc(100*time.Millisecond, cancel)
			rs = []geometryResponse{request("/api/optimize-network", f.Network, peer, ctx)}
			timer.Stop()
			cancel()
			rs = append(rs, network(false, peer))
		default:
			t.Fatal("unknown operation", p.Operation)
		}
		wall := time.Since(started).Seconds()
		cpu = capacityCPU() - cpu
		close(done)
		<-joined
		runtime.ReadMemStats(&after)
		cAfter := geometryCountersNow()
		if cAfter.RSS > rssMax {
			rssMax = cAfter.RSS
		}
		if cAfter.Current > currentMax {
			currentMax = cAfter.Current
		}
		if after.HeapAlloc > heapMax {
			heapMax = after.HeapAlloc
		}
		if len(limiter.slots) != 0 {
			t.Fatal("slots leaked")
		}
		totalBytes := 0
		slowest := 0.
		statuses := []int{}
		for i := range rs {
			if rs[i].Status != 200 && !(p.Operation == "cancel" && i == 0) {
				t.Fatalf("%s %s: %d %s", f.Domain, p.Operation, rs[i].Status, rs[i].Body)
			}
			totalBytes += rs[i].Bytes
			slowest = math.Max(slowest, rs[i].Wall)
			statuses = append(statuses, rs[i].Status)
			rs[i].Body = nil
		}
		jobSummary := map[string]any{}
		if job != nil {
			overlap := math.Max(0, math.Min(float64(rs[0].End.UnixNano()), float64(job.FinishedAt.UnixNano()))-math.Max(float64(rs[0].Start.UnixNano()), float64(job.StartedAt.UnixNano()))) / 1e9
			jobSummary = map[string]any{"status": job.Status, "cache_hit": job.CacheHit, "runs": job.CompletedRuns, "started_at": job.StartedAt, "finished_at": job.FinishedAt, "wall_seconds": job.FinishedAt.Sub(*job.StartedAt).Seconds(), "overlap_seconds": overlap}
			if overlap <= 0 {
				t.Fatal("no measured async overlap")
			}
		}
		row := map[string]any{"profile": profileName, "domain": f.Domain, "band": f.Band, "split": f.Split, "level": f.Level, "frequency_ghz": f.Frequency, "rays": f.Rays, "radius_m": f.Radius, "cells": n, "operation": p.Operation, "repeat": p.Repeat, "geometry": geo, "preflight_wall_seconds": preWall, "preflight_cpu_seconds": preCPUDelta, "preflight_allocation_bytes": preAfter.TotalAlloc - preBefore.TotalAlloc, "wall_seconds": wall, "cpu_seconds": cpu, "response_bytes": totalBytes, "deadline_headroom_seconds": 60 - slowest, "gc_count": after.NumGC - before.NumGC, "gc_pause_seconds": float64(after.PauseTotalNs-before.PauseTotalNs) / 1e9, "heap_before_bytes": before.HeapAlloc, "heap_after_bytes": after.HeapAlloc, "heap_sampled_max_bytes": heapMax, "allocation_bytes": after.TotalAlloc - before.TotalAlloc, "rss_before_bytes": cBefore.RSS, "rss_sampled_max_bytes": rssMax, "cgroup_before_bytes": cBefore.Current, "cgroup_sampled_max_bytes": currentMax, "cgroup_lifetime_peak_bytes": cAfter.Peak, "cgroup_cpu_seconds": float64(cAfter.CPU["usage_usec"]-cBefore.CPU["usage_usec"]) / 1e6, "throttled_seconds": float64(cAfter.CPU["throttled_usec"]-cBefore.CPU["throttled_usec"]) / 1e6, "throttled_periods": cAfter.CPU["nr_throttled"] - cBefore.CPU["nr_throttled"], "periods": cAfter.CPU["nr_periods"] - cBefore.CPU["nr_periods"], "memory_events": cAfter.Events, "work": raytracer.AuditCounts(), "interactive_slots_sampled_max": slotsMax, "async_active_sampled_max": asyncMax, "responses": rs, "async": jobSummary, "started_at": started}
		if e = encoder.Encode(row); e != nil {
			t.Fatal(e)
		}
		out.Sync()
		fmt.Printf("%s %d/%d %s %g %s %s r%d wall=%.3f cpu=%.3f\n", profileName, index+1, len(plans), f.Domain, f.Frequency, f.Level, p.Operation, p.Repeat, wall, cpu)
	}
}

func TestGeometryPreflightPrimitive(t *testing.T) {
	ring := []raytracer.Point{{Lon: 0, Lat: 0}, {Lon: .001, Lat: 0}, {Lon: .001, Lat: .001}, {Lon: 0, Lat: .001}, {Lon: 0, Lat: 0}}
	makePart := func(id, logical string, offset float64) *raytracer.BuildingFootprint {
		v := append([]raytracer.Point(nil), ring...)
		for i := range v {
			v[i].Lon += offset
		}
		b, _ := raytracer.BoundsFromPoints(v)
		return &raytracer.BuildingFootprint{ID: id, LogicalID: logical, Vertices: v, Bounds: b}
	}
	idx := raytracer.NewBuildingIndex([]*raytracer.BuildingFootprint{makePart("part1", "one", 0), makePart("part2", "one", .002), makePart("outside", "two", 1)})
	s := geometryDescribe(idx, raytracer.Bounds{MinLon: -.001, MinLat: -.001, MaxLon: .004, MaxLat: .004})
	if s.Footprints != 2 || s.Logical != 1 || s.Vertices != 10 || s.Edges != 8 || s.Max != 5 || s.Median != 5 {
		t.Fatalf("incorrect part/ring counts: %+v", s)
	}
	if s.Area <= 0 || s.VertexDensity <= 0 {
		t.Fatal("invalid domain area/density")
	}
	p := []raytracer.Point{{Lon: 32.8, Lat: 39.9}, {Lon: 32.9, Lat: 40}}
	small, large := geometryEnvelope(p, 100), geometryEnvelope(p, 800)
	if small.MinLon < large.MinLon || small.MaxLon > large.MaxLon || small.MinLat < large.MinLat || small.MaxLat > large.MaxLat {
		t.Fatal("larger envelope lost geometry")
	}
	if !geometryOverlap(small, large) {
		t.Fatal("nested envelopes should overlap")
	}
	empty := geometryDescribe(raytracer.EmptyBuildingIndex(), small)
	if empty.Footprints != 0 || empty.Vertices != 0 || empty.Edges != 0 {
		t.Fatal("empty index invented geometry")
	}
}
